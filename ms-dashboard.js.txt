import {supabase,requireUser,getProfile,wireCommon,esc} from "./ms-auth.js";
import {MS_STORAGE_BUCKET} from "./ms-config.js";
wireCommon();const user=await requireUser();let subjects=[],materials=[];
if(user){const profile=await getProfile(user.id),name=profile?.full_name||user.user_metadata?.full_name||user.email?.split("@")[0]||"Student";document.getElementById("user-name").textContent=name;document.getElementById("welcome-name").textContent=name.split(" ")[0];document.getElementById("user-email").textContent=user.email||"";document.getElementById("user-avatar").textContent=name[0].toUpperCase();if(profile?.role==="admin")document.getElementById("admin-nav").hidden=false;await Promise.all([loadSubjects(),loadMaterials(),loadCounts()]);}
async function loadSubjects(){const box=document.getElementById("subject-list"),r=await supabase.from("subjects").select("*").order("sort_order");if(r.error){box.innerHTML=`<p class="error-text">${esc(r.error.message)}</p>`;return;}subjects=r.data||[];document.getElementById("subject-status").textContent=subjects.length+" subjects";box.innerHTML=subjects.length?subjects.map((s,i)=>`<a class="subject-card" href="ms-subject.html?id=${encodeURIComponent(s.id)}"><span class="subject-card-icon">${["∑","◉","⌘","ϟ","⚙","🔧","♻"][i%7]}</span><div><b>${esc(s.name)}</b><small>${esc(s.hindi_name)}</small><p>${esc(s.description||"Units और resources खोलें")}</p></div><span class="card-arrow">↗</span></a>`).join(""):`<div class="empty-state">Subjects database में नहीं मिले। Setup SQL चलाएँ।</div>`;}
async function loadMaterials(){const box=document.getElementById("material-list"),r=await supabase.from("study_materials").select("title,description,file_name,file_path").eq("is_published",true).order("created_at",{ascending:false}).limit(8);if(r.error){box.textContent="Resources अभी load नहीं हुए: "+r.error.message;return;}materials=r.data||[];box.innerHTML=materials.length?materials.map(m=>`<a class="resource-item" href="#" data-path="${esc(m.file_path)}"><span class="file-icon">FILE</span><span><b>${esc(m.title)}</b><small>${esc(m.file_name||m.description)}</small></span><i>↓</i></a>`).join(""):`<div class="empty-state">अभी कोई resource publish नहीं हुआ है।</div>`;box.querySelectorAll("[data-path]").forEach(a=>a.onclick=async e=>{e.preventDefault();const {data,error}=await supabase.storage.from(MS_STORAGE_BUCKET).createSignedUrl(a.dataset.path,300);if(error)alert(error.message);else open(data.signedUrl,"_blank","noopener");});}
async function loadCounts(){const [b,n,p]=await Promise.all([supabase.from("bookmarks").select("id",{count:"exact",head:true}).eq("user_id",user.id),supabase.from("notes").select("id",{count:"exact",head:true}).eq("user_id",user.id),supabase.from("study_progress").select("id",{count:"exact",head:true}).eq("user_id",user.id).eq("completed",true)]);document.getElementById("stat-subjects").textContent=subjects.length||7;document.getElementById("stat-bookmarks").textContent=b.count??0;document.getElementById("stat-notes").textContent=n.count??0;document.getElementById("stat-units").textContent=p.count??0;}
document.getElementById("quick-note-form").onsubmit=async e=>{e.preventDefault();const out=document.getElementById("note-message");out.textContent="Saving…";const {error}=await supabase.from("notes").insert({user_id:user.id,title:document.getElementById("note-title").value.trim(),content:document.getElementById("note-content").value.trim()});out.textContent=error?error.message:"Note save हो गई ✓";if(!error){e.target.reset();loadCounts();}};
document.getElementById("global-search").oninput=()=>{const q=document.getElementById("global-search").value.toLowerCase().trim(),box=document.getElementById("search-results");if(!q){box.innerHTML="";return;}const hits=[...subjects.filter(s=>(s.name+" "+(s.hindi_name||"")).toLowerCase().includes(q)).map(s=>({title:s.name,type:"Subject",url:"ms-subject.html?id="+s.id})),...materials.filter(m=>(m.title+" "+(m.description||"")).toLowerCase().includes(q)).map(m=>({title:m.title,type:"Resource",url:"#"}))];box.innerHTML=hits.length?hits.map(h=>`<a href="${h.url}"><b>${esc(h.title)}</b><small>${h.type}</small></a>`).join(""):"<p class='muted'>कोई result नहीं मिला।</p>";};
if("serviceWorker"in navigator)addEventListener("load",()=>navigator.serviceWorker.register("ms-sw.js").catch(()=>{}));
