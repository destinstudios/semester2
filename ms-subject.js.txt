import {supabase,requireUser,wireCommon,esc} from "./ms-auth.js";import {MS_STORAGE_BUCKET} from "./ms-config.js";wireCommon();const user=await requireUser(),id=new URLSearchParams(location.search).get("id");if(user&&id)load();else if(user)document.getElementById("unit-list").textContent="Subject ID नहीं मिला।";
async function load(){const {data:s,error}=await supabase.from("subjects").select("*").eq("id",id).maybeSingle();if(error||!s){document.getElementById("subject-title").textContent="Subject नहीं मिला";return;}document.title=s.name+" | Mission Semester";document.getElementById("subject-title").textContent=s.name;document.getElementById("subject-description").textContent=s.hindi_name||s.description||"";const [u,q,m,p]=await Promise.all([supabase.from("units").select("*").eq("subject_id",id).order("sort_order"),supabase.from("questions").select("id,unit_id,question,answer,importance,sort_order").eq("subject_id",id).order("sort_order"),supabase.from("study_materials").select("*").eq("subject_id",id).eq("is_published",true).order("created_at",{ascending:false}),supabase.from("study_progress").select("unit_id,completed").eq("user_id",user.id)]);const units=u.data||[],qs=q.data||[],ms=m.data||[],done=new Set((p.data||[]).filter(x=>x.completed).map(x=>x.unit_id));document.getElementById("unit-count").textContent=units.length;document.getElementById("question-count").textContent=qs.length;document.getElementById("resource-count").textContent=ms.length;document.getElementById("unit-list").innerHTML=units.length?units.map(x=>`<article class="unit-card"><div><span class="unit-number">UNIT ${x.unit_number}</span><h3>${esc(x.title)}</h3><p>${esc(x.description||"")}</p></div><label class="progress-check"><input type="checkbox" data-unit="${x.id}" ${done.has(x.id)?"checked":""}> Completed</label></article>`).join(""):`<div class="empty-state">इस subject के units अभी add नहीं किए गए हैं। Admin Supabase Table Editor में units जोड़ सकता है।</div>`;document.querySelectorAll("[data-unit]").forEach(c=>c.onchange=async()=>{const {error}=await supabase.from("study_progress").upsert({user_id:user.id,unit_id:c.dataset.unit,completed:c.checked},{onConflict:"user_id,unit_id"});if(error){alert(error.message);c.checked=!c.checked;}});document.getElementById("question-list").innerHTML=qs.length?qs.map((x,i)=>`<details class="qa-item"><summary><span>Q${i+1}.</span> ${esc(x.question)} <small>${x.importance==="most_important"?"Most Important":x.importance==="important"?"Important":"Practice"}</small></summary><div class="qa-answer">${esc(x.answer||"Answer अभी add नहीं किया गया है।")}</div><button class="text-btn" data-bookmark="${x.id}">🔖 Bookmark</button></details>`).join(""):`<div class="empty-state">इस subject के questions अभी add नहीं किए गए हैं।</div>`;document.querySelectorAll("[data-bookmark]").forEach(b=>b.onclick=async()=>{const {error}=await supabase.from("bookmarks").insert({user_id:user.id,question_id:b.dataset.bookmark});b.textContent=error?(error.code==="23505"?"Already bookmarked":"Bookmark error"):"✓ Bookmarked";});const box=document.getElementById("subject-materials");box.innerHTML=ms.length?ms.map(x=>`<a class="resource-item" href="#" data-path="${esc(x.file_path)}"><span class="file-icon">FILE</span><span><b>${esc(x.title)}</b><small>${esc(x.description||x.file_name||"")}</small></span><i>↓</i></a>`).join(""):`<div class="empty-state">इस subject के लिए कोई published resource नहीं है।</div>`;box.querySelectorAll("[data-path]").forEach(a=>a.onclick=async e=>{e.preventDefault();const {data,error}=await supabase.storage.from(MS_STORAGE_BUCKET).createSignedUrl(a.dataset.path,300);if(error)alert(error.message);else open(data.signedUrl,"_blank","noopener");});}
