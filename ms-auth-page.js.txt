import {supabase} from "./ms-auth.js";
let mode="login";const form=document.getElementById("auth-form"),msg=document.getElementById("auth-message"),name=document.getElementById("full-name");
function setMode(m){mode=m;const s=m==="signup";document.getElementById("tab-login").classList.toggle("active",!s);document.getElementById("tab-signup").classList.toggle("active",s);document.getElementById("name-wrap").hidden=!s;name.required=s;document.getElementById("auth-title").textContent=s?"अपना account बनाएँ।":"वापस स्वागत है।";document.getElementById("auth-subtitle").textContent=s?"कुछ details भरें और learning शुरू करें।":"अपने account में login करें।";document.getElementById("auth-submit").textContent=s?"Create account →":"Login करें →";msg.textContent="";}
document.getElementById("tab-login").onclick=()=>setMode("login");document.getElementById("tab-signup").onclick=()=>setMode("signup");setMode("login");
form.addEventListener("submit",async e=>{e.preventDefault();const btn=document.getElementById("auth-submit");btn.disabled=true;msg.textContent="Please wait…";try{const email=document.getElementById("email").value.trim(),password=document.getElementById("password").value;if(mode==="signup"){const {data,error}=await supabase.auth.signUp({email,password,options:{data:{full_name:name.value.trim()},emailRedirectTo:"https://destinpresents.github.io/semester2/ms-dashboard.html"}});if(error)throw error;if(data.session)location.href="ms-dashboard.html";else msg.textContent="Account बन गया। Email provider की settings के अनुसार login करें।";}else{const {error}=await supabase.auth.signInWithPassword({email,password});if(error)throw error;location.href="ms-dashboard.html";}}catch(e){msg.textContent=e.message||"कुछ गलत हुआ।";}finally{btn.disabled=false;}});
